cmake_minimum_required(VERSION 3.10)
project(LoopForgePopolog LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(loopforge_logging INTERFACE)
target_include_directories(loopforge_logging INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(loopforge_logging INTERFACE Threads::Threads)

enable_testing()
add_executable(logging_test test/logging_test.cc test/logging_peer.cc)
target_link_libraries(logging_test PRIVATE loopforge_logging)
add_test(NAME logging COMMAND logging_test "${CMAKE_CURRENT_BINARY_DIR}/logging-test-output")
set_tests_properties(logging PROPERTIES TIMEOUT 60)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(echo_server source/echo/main.cc)
    add_executable(http_server source/http/main.cc)
    add_executable(recv_semantics_test test/recv_semantics_test.cc)
    foreach(target echo_server http_server recv_semantics_test)
        target_link_libraries(${target} PRIVATE loopforge_logging)
    endforeach()
    add_test(NAME recv_semantics COMMAND recv_semantics_test)
    file(COPY source/http/wwwroot DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
else()
    message(STATUS "Network examples require Linux; building logging tests only")
endif()
